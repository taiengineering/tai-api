## 🎯 실행 방법 (새 창에서)

### Python 코드 (Claude CTO 창에서 실행)

```python
import anthropic
import json

# Haiku 호출
client = anthropic.Anthropic()

rules_data = [
    {"id":"0002e3d4-39bc-415c-9c06-99c347773386","law":"화학물질관리법 제11조","summary":"화학물질 배출량조사에 필요한 자료를 제출해야 함"},
    {"id":"001cbcb5-972f-417e-9529-0c40162f09c9","law":"위험물안전관리법 시행규칙 제22조","summary":"완공검사 실시 시 완공검사업무대장을 작성하여 10년간 보관하여야 한다."},
    # ... (50개 모두)
]

condition_codes = ["annual_energy_toe", "boiler_capacity_kw", ..., "worker_count"]

prompt = f"""건설업 안전 규칙에 대해 가장 적절한 condition_code를 선택하세요.

## 사용 가능한 Condition Code (26개)
{', '.join(condition_codes)}

## 규칙 목록 (50개)
{"".join([f"\\n{i}. [{r['law']}] {r['summary']}" for i, r in enumerate(rules_data, 1)])}

## 응답 형식
[{{"id":"규칙ID","condition_code":"코드또는none"}},...]"""

message = client.messages.create(
    model="claude-3-5-haiku-20241022",
    max_tokens=2000,
    messages=[{"role": "user", "content": prompt}]
)

# 결과 파싱 및 Supabase UPDATE
```
